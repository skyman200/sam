// Code.gs : CBT 퀴즈 웹앱 (9주차 1강)
const QUIZ = 'Quiz';
const WRONG = '오답노트';

// 웹앱 주소를 열면 Index 화면을 보여 줍니다
function doGet() {
  return HtmlService
    .createHtmlOutputFromFile('Index')
    .setTitle('CBT 퀴즈')
    .addMetaTag('viewport',
    'width=device-width, initial-scale=1');
}

// 'Quiz' 탭의 문제를 모두 읽어 목록으로 돌려줍니다
function getQuiz() {
  const sh = SpreadsheetApp.getActive()
    .getSheetByName(QUIZ);
  if (!sh) throw new Error('Quiz 탭이 없습니다');
  return sh.getDataRange().getValues()
    .map((r, k) => (k > 0 && r[0] !== '')
      ? toItem(r, k + 1) : null)
    .filter(Boolean);
}

// 시트 한 줄을 문제 한 개로 바꿉니다
function toItem(r, row) {
  const ans = Number(r[5]);
  if (![1, 2, 3, 4].includes(ans)) {
    throw new Error(row + '행 F열은 1~4 숫자만');
  }
  const ch = r.slice(1, 5).map(String);
  return { q: String(r[0]), ch: ch,
    ans: ans, exp: String(r[6]) };
}

// 틀린 문제를 '오답노트' 탭에 한 줄 저장합니다
function saveWrong(i, pick) {
  const ss = SpreadsheetApp.getActive();
  const sh = ss.getSheetByName(WRONG)
    || ss.insertSheet(WRONG);
  const it = getQuiz()[i];
  if (!it) throw new Error('없는 문제 번호');
  sh.appendRow([new Date(), it.q,
    pick, it.ans, it.exp]);
  return true;
}

// 파일 종류(mime)를 알아냅니다. 모르면 빈 글자
function mimeOf(f) {
  if (/\.pdf$/i.test(f.name)
    || f.type === 'application/pdf') return 'application/pdf';
  if (/\.txt$/i.test(f.name)
    || f.type === 'text/plain') return 'text/plain';
  return /^image\/(png|jpeg|webp|heic|heif)$/
    .test(f.type) ? f.type : '';
}

const QUIZ_MODEL = 'gemini-3.5-flash-lite';
const QUIZ_ASK = '이 유인물로 4지선다 10문제.'

// 웹앱이 보낸 파일로 문제를 만들어 시트에 씁니다
function makeQuizFromFile(b64, mime) {
  const sh = SpreadsheetApp.getActive()
    .getSheetByName(QUIZ);
  if (!sh) throw new Error('Quiz 탭이 없습니다');
  const rows = askQuiz_({ inline_data: {
    mime_type: mime, data: b64 } }, QUIZ_ASK);
  if (rows.length < 5) throw new Error('쓸 만한'
    + ' 문제가 ' + rows.length + '개뿐입니다.'
    + ' 다시 눌러 주세요');
  // 글 앞에 ' 를 붙여 '2:1' 같은 글이 시간으로
  // 바뀌지 않게 글자 그대로 적습니다
  const safe = rows.map(r => r.map(v =>
    typeof v === 'string' ? "'" + v : v));
  sh.getRange('A2:G').clearContent();
  sh.getRange(2, 1, safe.length, 7)
    .setValues(safe);
  return safe.length;
}

// AI에게 묻고, 시트에 쓸 줄(7칸) 목록을 받습니다
function askQuiz_(part, ask) {
  const key = PropertiesService
    .getScriptProperties()
    .getProperty('GEMINI_API_KEY');
  if (!key) throw new Error('GEMINI_API_KEY 없음'
    + ': 프로젝트 설정 → 스크립트 속성');
  const url = 'https://generativelanguage'
    + '.googleapis.com/v1beta/models/'
    + QUIZ_MODEL + ':generateContent';
  const s = { type: 'STRING' };
  const item = { type: 'OBJECT', properties: {
    q: s, ch: { type: 'ARRAY', items: s,
      minItems: 4, maxItems: 4 },
    ans: { type: 'INTEGER', minimum: 1,
      maximum: 4, description: '첫 보기가 1' },
    exp: s },
    required: ['q', 'ch', 'ans', 'exp'],
    propertyOrdering: ['q', 'ch', 'ans', 'exp'] };
  const body = { contents: [{ parts: [
    part, { text: ask }] }],
    generationConfig: {
      responseMimeType: 'application/json',
      responseSchema: { type: 'ARRAY',
        items: item, maxItems: 10 } } };
  const opt = { method: 'post',
    contentType: 'application/json',
    headers: { 'x-goog-api-key': key },
    payload: JSON.stringify(body),
    muteHttpExceptions: true };
  let res; // 바쁘면(429·503) 3초 뒤 한 번 더
  for (let t = 0; t < 2; t++) {
    res = UrlFetchApp.fetch(url, opt);
    const c0 = res.getResponseCode();
    if (![429, 500, 503].includes(c0)) break;
    Utilities.sleep(3000);
  }
  const code = res.getResponseCode();
  const json = res.getContentText();
  if (code !== 200) {
    let m = json;
    try { m = JSON.parse(json).error.message; }
    catch (e) {}
    throw new Error('오류 ' + code + ' : '
      + String(m).slice(0, 120));
  }
  return readQuiz_(JSON.parse(json));
}

// AI 답에서 문제 목록을 꺼내고, 이상한 문제는 뺍니다
function readQuiz_(data) {
  const c = (data.candidates || [])[0];
  const ps = c && c.content && c.content.parts;
  const text = (ps || []).filter(
    p => p.text && !p.thought)
    .map(p => p.text).join('');
  if (!text) throw new Error('AI가 답을 주지'
    + ' 않았습니다. 다른 파일로 해 보세요');
  let list;
  try { list = JSON.parse(text); }
  catch (e) { list = null; }
  if (!Array.isArray(list)) throw new Error(
    'AI 답이 중간에 끊겼습니다. 다시 눌러 주세요');
  return list.map(toRow_).filter(r => r)
    .slice(0, 10);
}

// 문제 한 개를 검사하고 시트 한 줄로 바꿉니다
function toRow_(it) {
  const ch = Array.isArray(it.ch) ? it.ch : [];
  const ans = Number(it.ans);
  const r = [it.q].concat(ch, ans, it.exp);
  const ok = ch.length === 4
    && [1, 2, 3, 4].includes(ans) && r.every(
      v => v != null && String(v).trim() !== '');
  return ok ? r : null;
}

  const rows = sh.getDataRange().getValues()
    .filter(function (r) {
      return r[1] !== '' && r[0] !== '날짜';
    });
  if (rows.length === 0) {
    throw new Error('오답노트에 기록이 없습니다');
  }
  return rows;
}

const REMIND_HOUR = 21;

// 오늘 틀린 문제 개수를 내 메일로만 보냅니다
function sendRemind() {
  const today = new Date().toDateString();
  const list = wrongRows_().filter(r =>
    new Date(r[0]).toDateString() === today);
  if (list.length === 0) {
    Logger.log('오늘 틀린 문제가 없습니다');

function makeRemindTrigger() {
  const all = ScriptApp.getProjectTriggers();
  for (const t of all) {
    const name = t.getHandlerFunction();
    if (name === 'sendRemind') {
      ScriptApp.deleteTrigger(t);
    }
  }

  ScriptApp.newTrigger('sendRemind')
    .timeBased()
    .everyDays(1)
    .atHour(REMIND_HOUR)
    .create();
  Logger.log('매일 알림 트리거를 만들었습니다');
}

// Result.gs : 웹앱 결과 화면에 보낼 값 만들기
// Code·Score·Notes·Exam 파일이 먼저 필요합니다
// 필요 권한: 스프레드시트, 문서(Drive), 캘린더

// 점수·정답률·틀린 문제·TOP3 를 객체로 돌려줍니다
function getResult() {
  const total = getQuiz().length;
  const last = {}, cnt = {};
  wrongRows_().forEach(r => {
    last[r[1]] = r;
    cnt[r[1]] = (cnt[r[1]] || 0) + 1;
  });
  const qs = Object.keys(last);
  const wrongs = qs.map(q => ({ q: q,
    mine: Number(last[q][2]),
    answer: Number(last[q][3]),
    why: String(last[q][4]) }));
  const top3 = qs.slice()
    .sort((a, b) => cnt[b] - cnt[a])
    .slice(0, 3)
    .map(q => ({ q: q, n: cnt[q] }));
  const score = Math.max(
    0, total - qs.length);
  const rate = total
    ? Math.round(score / total * 100) : 0;
  return { score: score, total: total,
    rate: rate, wrongs: wrongs, top3: top3 };
}
