// 스크랩 웹앱 (7주차 1강)
const SHEET_NAME = '스크랩';
const LIST_MAX = 20;

// ?page= 이름의 화면을 보여 줍니다(없으면 Index)
function doGet(e) {
  const page = (e && e.parameter.page)
    || 'Index';
  let out;
  try {
    out = HtmlService
      .createHtmlOutputFromFile(page);
  } catch (err) {
    out = HtmlService
      .createHtmlOutputFromFile('Index');
  }
  return out.setTitle('내 스크랩')
    .addMetaTag('viewport',
    'width=device-width, initial-scale=1');
}

// '스크랩' 탭을 찾아 돌려줍니다(없으면 오류)
function getSheet_() {
  const sheet = SpreadsheetApp
    .getActiveSpreadsheet()
    .getSheetByName(SHEET_NAME);
  if (!sheet) {
    throw new Error('스크랩 탭이 없습니다');
  }
  return sheet;
}

// 주제·링크·인용구와 오늘 날짜를 한 줄 저장합니다
function addScrap(topic, link, quote) {
  if (!topic || !link) {
    throw new Error('주제와 링크를 넣어 주세요');
  }
  getSheet_().appendRow(
    [topic, link, quote, new Date()]);
  return '저장했습니다';
}

// 스크랩 최신 20건을 화면용 목록으로 돌려줍니다
function listScraps() {
  const rows = getSheet_().getDataRange()
    .getValues().slice(1)
    .filter(function (r) { return r[0]; });
  const zone = Session.getScriptTimeZone();

  return rows.reverse().slice(0, LIST_MAX)
    .map(function (r) {
      const d = r[3];
      return {
        topic: String(r[0]),
        link: String(r[1]),
        quote: String(r[2]),
        date: d instanceof Date
          ? Utilities.formatDate(d, zone,
            'yyyy-MM-dd HH:mm') : String(d)
      };
    });
}

// 스크랩 목록을 내 메일 주소로만 보냅니다
function mailList() {
  const rows = getSheet_().getDataRange()
    .getValues().slice(1);
  if (rows.length === 0) {
    return '보낼 스크랩이 없습니다';
  }

  let body = '';
  rows.forEach(function (r, i) {
    body += (i + 1) + '. ' + r[0] + '\n'
      + r[1] + '\n' + r[2] + '\n\n';
  });

  const me = Session.getEffectiveUser()
    .getEmail();
  MailApp.sendEmail(me,
    '[스크랩] 내 자료 ' + rows.length + '건',
    body);
  return me + ' 주소로 보냈습니다';
}

// Feed.gs : 새 글 목록(RSS)을 읽어 스크랩에 추가
// Code.gs의 getSheet_ 를 쓰므로
// Code.gs가 먼저 있어야 합니다
const FEED_URL = 'https://sam.ptclinical'
  + '.workers.dev/practice/feed.xml';
const FEED_COUNT = 5;

// 최신 5건의 제목·링크를 '스크랩' 탭 끝에 추가합니다
function importFeed() {
  const items = readFeed_()
    .slice(0, FEED_COUNT);
  const sheet = getSheet_();
  const known = savedLinks_(sheet);

  let added = 0;
  items.forEach(function (item) {
    if (known.includes(item.link)) {
      return; // 이미 저장한 링크는 건너뜁니다
    }
    sheet.appendRow([item.title,
      item.link, '', new Date()]);
    added++;
  });

  Logger.log(added + '건 추가했습니다');
  return added;
}

// 주소에서 RSS를 받아 제목·링크 목록으로 바꿉니다
function readFeed_() {
  const res = UrlFetchApp.fetch(FEED_URL,
    { muteHttpExceptions: true });
  const code = res.getResponseCode();
  if (code !== 200) {
    throw new Error('피드 오류 ' + code);
  }

  const xml = res.getContentText('UTF-8');
  const root = XmlService.parse(xml)
    .getRootElement();
  const channel = root.getChild('channel');
  if (!channel) {
    throw new Error('RSS 형식이 아닙니다');
  }
  return channel.getChildren('item')
    .map(function (e) {
      return {
        title: e.getChildText('title'),
        link: e.getChildText('link')
      };
    });
}

// 시트에 이미 있는 링크(B열)를 목록으로 모읍니다
function savedLinks_(sheet) {
  const last = sheet.getLastRow();
  if (last < 2) {
    return [];
  }
  return sheet.getRange(2, 2, last - 1, 1)
    .getValues()
    .map(function (r) {
      return String(r[0]);
    });
}

// Report.gs : 스크랩 현황을 요약해 매주 메일로 보냄
// Code.gs의 getSheet_ 를 쓰므로
// Code.gs가 먼저 있어야 합니다
const REPORT_HOUR = 8;
const WEEK_MS = 7 * 24 * 60 * 60 * 1000;

// 스크랩 개수와 최근 저장일을 내 메일로 보냅니다
function weeklyReport() {
  const rows = getSheet_().getDataRange()
    .getValues().slice(1)
    .filter(function (r) { return r[0]; });
  const dates = pickDates_(rows);
  const zone = Session.getScriptTimeZone();

  let latest = '날짜 없음';
  if (dates.length > 0) {
    const last = new Date(
      Math.max.apply(null, dates));
    latest = Utilities.formatDate(
      last, zone, 'yyyy-MM-dd');
  }

  const weekAgo = new Date(
    Date.now() - WEEK_MS);
  const recent = dates.filter(function (d) {
    return d >= weekAgo;
  });
  const today = Utilities.formatDate(
    new Date(), zone, 'yyyy-MM-dd');

  const body = '전체 스크랩: '
    + rows.length + '건\n'
    + '최근 7일: ' + recent.length + '건\n'
    + '가장 최근 저장일: ' + latest + '\n';
  const me = Session.getEffectiveUser()
    .getEmail();
  MailApp.sendEmail(me,
    '[주간 보고] ' + today, body);
  Logger.log(me + ' 주소로 보냈습니다');
}

// 매주 월요일 아침 weeklyReport 가 돌게 예약합니다
function makeWeeklyTrigger() {
  const all = ScriptApp.getProjectTriggers();
  for (const t of all) {
    const name = t.getHandlerFunction();
    if (name === 'weeklyReport') {
      ScriptApp.deleteTrigger(t);
    }
  }

  ScriptApp.newTrigger('weeklyReport')
    .timeBased()
    .onWeekDay(ScriptApp.WeekDay.MONDAY)
    .atHour(REPORT_HOUR)
    .create();
  Logger.log('주간 트리거를 만들었습니다');
}

// 각 줄의 날짜(D열)만 모읍니다(잘못된 값은 뺌)
function pickDates_(rows) {
  return rows
    .map(function (r) {
      return new Date(r[3]);
    })
    .filter(function (d) {
      return !isNaN(d.getTime());
    });
}

// Clean.gs : 스크랩 중복 지우기와 주제별 개수
// Code.gs 의 getSheet_ 가 먼저 필요합니다
// 필요 권한: 구글 스프레드시트 읽기·쓰기

// 같은 링크가 또 나오면 뒤쪽 줄을 지웁니다
function removeDuplicates() {
  const sheet = getSheet_();
  const all = sheet.getDataRange()
    .getValues();
  const dups = dupRows_(all);

  dups.forEach(function (n) {
    sheet.deleteRow(n);
  });
  Logger.log(dups.length + '개 줄을 지웠습니다');
  return dups.length;
}

// 앞에서 이미 본 링크의 줄 번호를 모읍니다
function dupRows_(rows) {
  const seen = {};
  const found = [];

  for (let i = 1; i < rows.length; i++) {
    const link = String(rows[i][1]).trim();
    if (seen[link]) found.push(i + 1);
    else if (link) seen[link] = true;
  }
  return found.reverse(); // 아래 줄부터 지웁니다
}

// 화면용: 중복을 지우고 주제별 개수와 함께 돌려줍니다
function cleanUp() {
  const removed = removeDuplicates();
  return { removed: removed,
    counts: countByTopic() };
}

// 주제별로 스크랩이 몇 개인지 세어 돌려줍니다
function countByTopic() {
  const rows = getSheet_().getDataRange()
    .getValues().slice(1);
  const count = {};

  rows.forEach(function (r) {
    if (r[0] === '') return;
    count[r[0]] = (count[r[0]] || 0) + 1;
  });
  return Object.keys(count).map(
    function (k) {
      Logger.log(k + ': ' + count[k] + '개');
      return { topic: k, n: count[k] };
    });
}

// Code.gs 의 getSheet_ 가 먼저 필요합니다
// 필요 권한: 스프레드시트 읽기, 구글 문서 만들기
const SHARE_TITLE = '스크랩 모음';

// 주제 순서대로 스크랩을 문서 한 개에 담습니다
function exportToDoc() {
  const rows = getSheet_().getDataRange()
    .getValues().slice(1)
    .filter(function (r) { return r[0]; });
  if (rows.length === 0) {
    throw new Error('보낼 스크랩이 없습니다');
  }

  rows.sort(function (a, b) {
    return String(a[0]).localeCompare(
      String(b[0]));
  });

  const doc = DocumentApp.create(
    SHARE_TITLE);
  rows.forEach(function (r) {
    addItem_(doc.getBody(), r);
  });
  const url = doc.getUrl();
  doc.saveAndClose();
  Logger.log('문서 완성: ' + url);
  return url;
}

// 문서에 스크랩 한 건(주제·인용구·링크)을 적습니다
function addItem_(body, r) {
  const H = DocumentApp.ParagraphHeading;
  body.appendParagraph(String(r[0]))
    .setHeading(H.HEADING2);
  body.appendParagraph(String(r[2]));
  body.appendParagraph(String(r[1]))
    .editAsText().setLinkUrl(String(r[1]));
}

// 화면용: 문서를 만들고 이름·링크·건수를 돌려줍니다
function exportDoc() {
  const url = exportToDoc();
  let n = 0;
  countByTopic().forEach(function (c) {
    n += c.n;
  });
  return { url: url, title: SHARE_TITLE,
    n: n };
}
