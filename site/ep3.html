const QUIZ = 'Quiz';
const WRONG = '오답노트';

// 웹앱 주소를 열면 Index 화면을 보여 줍니다
function doGet() {
  return HtmlService
    .createHtmlOutputFromFile('Index')
    .setTitle('CBT 퀴즈')
    .addMetaTag('viewport',
    'width=device-width, initial-scale=1');
}

// 'Quiz' 탭의 문제를 모두 읽어 목록으로 돌려줍니다
function getQuiz() {
  const sh = SpreadsheetApp.getActive()
    .getSheetByName(QUIZ);
  if (!sh) throw new Error('Quiz 탭이 없습니다');
  return sh.getDataRange().getValues()
    .map((r, k) => (k > 0 && r[0] !== '')
      ? toItem(r, k + 1) : null)
    .filter(Boolean);
}

// 시트 한 줄을 문제 한 개로 바꿉니다
function toItem(r, row) {
  const ans = Number(r[5]);
  if (![1, 2, 3, 4].includes(ans)) {
    throw new Error(row + '행 F열은 1~4 숫자만');
  }
  const ch = r.slice(1, 5).map(String);
  return { q: String(r[0]), ch: ch,
    ans: ans, exp: String(r[6]) };
}

// 틀린 문제를 '오답노트' 탭에 한 줄 저장합니다
function saveWrong(i, pick) {
  const ss = SpreadsheetApp.getActive();
  const sh = ss.getSheetByName(WRONG)
    || ss.insertSheet(WRONG);
  const it = getQuiz()[i];
  if (!it) throw new Error('없는 문제 번호');
  sh.appendRow([new Date(), it.q,
    pick, it.ans, it.exp]);
  return true;
}

// Exam.gs : 시험일 D-14, D-7, D-1 을 캘린더에 등록
// 새 파일로 만듭니다. Code.gs와 이름이 겹치지 않습니다
const EXAM = '시험';
const DAYS = [14, 7, 1];

// '시험' 탭 2행으로 일정을 만들고 글자 목록을 돌려줍니다
function addExam() {
  const sh = SpreadsheetApp.getActive()
    .getSheetByName(EXAM);
  if (!sh) throw new Error('시험 탭이 없습니다');
  const name = sh.getRange('A2').getValue();
  const day = sh.getRange('B2').getValue();
  if (!name) throw new Error('A2에 과목명을 쓰세요');
  if (!(day instanceof Date)) {
    throw new Error('B2는 2026-12-14처럼 날짜로');
  }

  const c = CalendarApp.getDefaultCalendar();
  const zone = Session.getScriptTimeZone();
  return DAYS.map(n => {
    const d = new Date(day.getTime());
    d.setDate(d.getDate() - n);
    const t = 'D-' + n + ' ' + name;
    addOnce(c, t, d);
    const ymd = Utilities.formatDate(
      d, zone, 'yyyy-MM-dd');
    return t + ' · ' + ymd;
  });
}

// 같은 제목이 그날 없을 때만 종일 일정을 만듭니다
function addOnce(c, title, d) {
  const old = c.getEventsForDay(d)
    .filter(e => e.getTitle() === title);
  if (old.length > 0) {
    console.log(title + ' 이미 있음, 건너뜀');
    return;
  }
  c.createAllDayEvent(title, d);
  console.log(title + ' 등록함');
}

    body.appendParagraph('해설: ' + r[4]);
  });

  const url = doc.getUrl();
  doc.saveAndClose();
  Logger.log('복습 자료 완성: ' + url);
  return url;

// 오답노트 탭에서 틀린 문제 줄만 읽어 옵니다
function readWrong_() {
  const sh = SpreadsheetApp.getActive()
    .getSheetByName(WRONG);
  if (!sh) {
    throw new Error('오답노트 탭이 없습니다');
  }
  const rows = sh.getDataRange().getValues()
    .filter(function (r) {
      return r[1] !== '' && r[0] !== '날짜';

  if (rows.length === 0) {

// Score.gs : 퀴즈 점수 기록과 약한 문제 찾기
// Code.gs 의 WRONG·getQuiz 가 먼저 필요합니다
// 필요 권한: 구글 스프레드시트 읽기·쓰기
const SCORE_TAB = '점수';

// 오늘 틀린 수로 점수·정답률을 적습니다(하루 1회)
function saveScore() {
  const today = new Date().toDateString();
  const wrong = wrongRows_().filter(r =>
    new Date(r[0]).toDateString() === today);
  const total = getQuiz().length;
  const right = total - wrong.length;

  const rate = Math.round(
    right / total * 100);
  const ss = SpreadsheetApp.getActive();
  const sh = ss.getSheetByName(SCORE_TAB)
    || ss.insertSheet(SCORE_TAB);
  sh.appendRow(
    [new Date(), right, total, rate]);
  Logger.log('정답률 ' + rate + '%');
}

// 가장 자주 틀린 문제 3개를 보여 줍니다
function showWeakTop3() {
  const count = {};
  wrongRows_().forEach(r => {
    count[r[1]] = (count[r[1]] || 0) + 1;
  });

  const top = Object.keys(count)
    .sort((a, b) => count[b] - count[a])
    .slice(0, 3);
  top.forEach((q, i) => Logger.log(
    (i + 1) + '위 ' + count[q] + '번: ' + q));
}

// 오답노트의 틀린 문제 줄을 읽습니다(없으면 빈 목록)
function wrongRows_() {
  const sh = SpreadsheetApp.getActive()
    .getSheetByName(WRONG);
  if (!sh) return [];
  return sh.getDataRange().getValues()
    .filter(r => r[1] && r[0] !== '날짜');
}

// 점수·정답률·틀린 문제·TOP3 를 객체로 돌려줍니다
function getResult() {
  const total = getQuiz().length;
  const last = {}, cnt = {};
  wrongRows_().forEach(r => {
    last[r[1]] = r;
    cnt[r[1]] = (cnt[r[1]] || 0) + 1;
  });
  const qs = Object.keys(last);
  const wrongs = qs.map(q => ({ q: q,
    mine: Number(last[q][2]),
    answer: Number(last[q][3]),
    why: String(last[q][4]) }));
  const top3 = qs.slice()
    .sort((a, b) => cnt[b] - cnt[a])
    .slice(0, 3)
    .map(q => ({ q: q, n: cnt[q] }));
  const score = Math.max(
    0, total - qs.length);
  const rate = total
    ? Math.round(score / total * 100) : 0;
  return { score: score, total: total,
    rate: rate, wrongs: wrongs, top3: top3 };
}
