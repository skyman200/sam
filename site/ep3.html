  4. 함수를 골라 실행을 누르고, 권한 창에서 안내대로 진행한다.

// Code.gs : CBT 퀴즈 웹앱 (9주차 1강)
const QUIZ = 'Quiz';
const WRONG = '오답노트';

// 웹앱 주소를 열면 Index 화면을 보여 줍니다
function doGet() {
  return HtmlService
    .createHtmlOutputFromFile('Index')
    .setTitle('CBT 퀴즈')
    .addMetaTag('viewport',
    'width=device-width, initial-scale=1');
}

// 'Quiz' 탭의 문제를 모두 읽어 목록으로 돌려줍니다
function getQuiz() {
  const sh = SpreadsheetApp.getActive()
    .getSheetByName(QUIZ);
  if (!sh) throw new Error('Quiz 탭이 없습니다');
  return sh.getDataRange().getValues()
    .map((r, k) => (k > 0 && r[0] !== '')
      ? toItem(r, k + 1) : null)
    .filter(Boolean);
}

// 시트 한 줄을 문제 한 개로 바꿉니다
function toItem(r, row) {
  const ans = Number(r[5]);
  if (![1, 2, 3, 4].includes(ans)) {
    throw new Error(row + '행 F열은 1~4 숫자만');
  }
  const ch = r.slice(1, 5).map(String);
  return { q: String(r[0]), ch: ch,
    ans: ans, exp: String(r[6]) };
}

// 틀린 문제를 '오답노트' 탭에 한 줄 저장합니다
function saveWrong(i, pick) {
  const ss = SpreadsheetApp.getActive();
  const sh = ss.getSheetByName(WRONG)
    || ss.insertSheet(WRONG);
  const it = getQuiz()[i];
  if (!it) throw new Error('없는 문제 번호');
  sh.appendRow([new Date(), it.q,
    pick, it.ans, it.exp]);
  return true;
}

const $ = id => document.getElementById(id);
let quiz = [], n = 0, score = 0;
google.script.run
  .withSuccessHandler(start)
  .withFailureHandler(fail)
  .getQuiz();

// 오류가 나면 문제 자리에 이유를 보여 줍니다
function fail(e) {
  $('q').textContent = '오류: ' + e.message;
}
// 문제를 받으면 첫 문제부터 시작합니다
function start(data) {
  quiz = data;
  if (quiz.length > 0) return show();
  fail({ message: 'Quiz 탭에 문제가 없습니다' });
}

// 지금 문제와 보기 버튼을 화면에 그립니다
function show() {
  const it = quiz[n];
  $('q').textContent = (n + 1) + '. ' + it.q;
  $('msg').textContent = '';
  $('go').style.display = 'none';
  $('ch').innerHTML = '';
  it.ch.forEach((c, k) => addBtn(c, k + 1));
}

// 보기 버튼 하나를 만들어 붙입니다
function addBtn(text, no) {
  const b = document.createElement('button');
  b.textContent = no + ') ' + text;
  b.onclick = () => pick(no);
  $('ch').appendChild(b);
}

// 보기를 누르면 맞았는지 알려 줍니다
function pick(no) {
  const it = quiz[n];
  document.querySelectorAll('#ch button')
    .forEach(b => { b.disabled = true; });
  if (no === it.ans) {
    score++;
    $('msg').textContent = '정답. ' + it.exp;
  } else { wrong(it, no); }
  $('go').style.display = 'block';
}

// 다음 문제로 넘어가고, 끝나면 점수를 보여 줌
function goNext() {
  n++;
  if (n < quiz.length) return show();
  $('q').textContent = '점수 ' + score
    + ' / ' + quiz.length;
  $('ch').innerHTML = '';
  $('msg').textContent = '틀린 문제는 오답노트에';
  $('go').style.display = 'none';
}

// 오답이면 정답을 알리고 시트에 저장합니다
function wrong(it, no) {
  $('msg').textContent = '오답. 정답은 '
    + it.ans + '번. ' + it.exp;
  google.script.run.withFailureHandler(fail)
    .saveWrong(n, no);
}
</script>

// Exam.gs : 시험일 D-14, D-7, D-1 을 캘린더에 등록
// 새 파일로 만듭니다. Code.gs와 이름이 겹치지 않습니다
const EXAM = '시험';
const DAYS = [14, 7, 1];

// '시험' 탭 2행의 과목·시험일로 일정을 만듭니다
function addExam() {
  const sh = SpreadsheetApp.getActive()
    .getSheetByName(EXAM);
  if (!sh) throw new Error('시험 탭이 없습니다');
  const name = sh.getRange('A2').getValue();
  const day = sh.getRange('B2').getValue();
  if (!name) throw new Error('A2에 과목명을 쓰세요');
  if (!(day instanceof Date)) {
    throw new Error('B2는 2026-12-14처럼 날짜로');
  }

  const c = CalendarApp.getDefaultCalendar();
  DAYS.forEach(n => {
    const d = new Date(day.getTime());
    d.setDate(d.getDate() - n);
    addOnce(c, 'D-' + n + ' ' + name, d);
  });
}

// 같은 제목이 그날 없을 때만 종일 일정을 만듭니다
function addOnce(c, title, d) {
  const old = c.getEventsForDay(d)
    .filter(e => e.getTitle() === title);
  if (old.length > 0) {
    console.log(title + ' 이미 있음, 건너뜀');
    return;
  }
  c.createAllDayEvent(title, d);
  console.log(title + ' 등록함');
}

// Notes.gs : 오답노트 탭을 복습 자료 문서로 만들기
// Code.gs의 WRONG 상수를 쓰므로 Code.gs가 먼저 있어야 합니다
const REVIEW_TITLE = '복습 자료';

// 오답노트 전체를 '복습 자료' 문서 하나로 만듭니다
function makeReviewDoc() {
  const rows = readWrong_();
  const zone = Session.getScriptTimeZone();
  const today = Utilities.formatDate(
    new Date(), zone, 'yyyy-MM-dd');

  const doc = DocumentApp.create(
    REVIEW_TITLE + ' ' + today);
  const body = doc.getBody();
  const H = DocumentApp.ParagraphHeading;
  body.getParagraphs()[0]
    .setText(REVIEW_TITLE)
    .setHeading(H.HEADING1);
  body.appendParagraph('틀린 문제 '
    + rows.length + '개 (' + today + ')');

  rows.forEach(function (r, i) {
    body.appendParagraph(
      (i + 1) + '. ' + r[1])
      .setHeading(H.HEADING3);
    body.appendParagraph('내 답: ' + r[2]
      + '번 / 정답: ' + r[3] + '번');
    body.appendParagraph('해설: ' + r[4]);
  });

  const url = doc.getUrl();
  doc.saveAndClose();
  Logger.log('복습 자료 완성: ' + url);
  return url;
}

// 오답노트 탭에서 틀린 문제 줄만 읽어 옵니다
function readWrong_() {
  const sh = SpreadsheetApp.getActive()
    .getSheetByName(WRONG);
  if (!sh) {
    throw new Error('오답노트 탭이 없습니다');
  }
  const rows = sh.getDataRange().getValues()
    .filter(function (r) {
      return r[1] !== '' && r[0] !== '날짜';
    });
  if (rows.length === 0) {
    throw new Error('오답노트에 기록이 없습니다');
  }
  return rows;
}
