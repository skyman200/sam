// Summary.gs : 셀에서 쓰는 AI 요약 함수
// 모델 이름은 자주 바뀝니다. 바뀌면 여기만
const MODEL = 'gemini-flash-latest';

/**
 * 글을 3줄로 요약합니다.
 * @customfunction
 */
function AI_SUMMARY(text) {
  if (!text) return '';

  const key = PropertiesService
    .getScriptProperties()
    .getProperty('GEMINI_API_KEY');
  if (!key) {
    throw new Error('GEMINI_API_KEY 없음');
  }
  const url = 'https://generativelanguage'
    + '.googleapis.com/v1beta/models/'
    + MODEL + ':generateContent';

  const body = { contents: [{ parts: [{
    text: '대학생 시험 대비용으로 별표 없이 '
      + '3줄로 요약해 줘:\n' + text
  }] }] };
  const res = UrlFetchApp.fetch(url, {
    method: 'post',
    contentType: 'application/json',
    headers: { 'x-goog-api-key': key },
    payload: JSON.stringify(body),
    muteHttpExceptions: true });

  const code = res.getResponseCode();
  const json = res.getContentText();
  if (code !== 200) {
    throw new Error('오류 ' + code + ' : '
      + json.slice(0, 80));
  }
  const c = JSON.parse(json).candidates;
  if (!c) throw new Error('답이 비었습니다');
  return c[0].content.parts[0].text;
}

// SamSummary.gs : 키 없이 쓰는 수업용 요약 함수
// 수업용 서버(Cloudflare)에 글을 보내 3줄로 받습니다
const SAM_URL = 'https://sam.ptclinical'
  + '.workers.dev/api/summarize';
const SAM_CODE = 'sam-2026';
const SAM_MIN_LENGTH = 20;
const SAM_CACHE_SECONDS = 21600; // 6시간

/**
 * 키 없이 글을 3줄로 요약합니다.
 * @customfunction
 */
function SAM_SUMMARY(text) {
  if (Array.isArray(text)) {
    text = text.join(' ');
  }
  text = String(text || '').trim();
  if (text.length < SAM_MIN_LENGTH) {
    return '20자 이상 글을 넣어 주세요';
  }

  // 같은 글은 6시간 동안 저장해 둔 답을 씁니다
  const cache = CacheService
    .getScriptCache();
  const cacheKey = makeCacheKey_(text);
  const saved = cache.get(cacheKey);
  if (saved) {
    return saved;
  }

  const res = UrlFetchApp.fetch(SAM_URL, {
    method: 'post',
    contentType: 'application/json',
    headers: { 'x-sam-code': SAM_CODE },
    payload: JSON.stringify({ text: text }),
    muteHttpExceptions: true });

  const code = res.getResponseCode();
  const json = res.getContentText();
  if (code !== 200) {
    throw new Error('오류 ' + code + ' : '
      + json.slice(0, 80));
  }
  const summary = JSON.parse(json).summary;
  if (!summary) throw new Error('답이 비었습니다');
  cache.put(cacheKey, summary,
    SAM_CACHE_SECONDS);
  return summary;
}

// 글 내용으로 짧은 저장 이름(캐시 키)을 만듭니다
function makeCacheKey_(text) {
  const bytes = Utilities.computeDigest(
    Utilities.DigestAlgorithm.MD5,
    text, Utilities.Charset.UTF_8);
  return 'sam_'
    + Utilities.base64EncodeWebSafe(bytes);
}

  Logger.log(out);
}

// WebSummary.gs : 요약 노트 화면용 함수
// SamSummary.gs·Pdf.gs 가 먼저 필요합니다
// 필요 권한: 스프레드시트 읽기·쓰기, 외부 연결, 드라이브
const NOTE_TAB = '필기';
const NOTE_MAX = 10;

// '필기' 탭 A열(2행부터)을 {row, text} 로 돌려줍니다
function loadNotes() {
  const sheet = SpreadsheetApp
    .getActiveSpreadsheet()
    .getSheetByName(NOTE_TAB);
  if (!sheet) throw new Error('필기 탭이 없습니다');

  const last = sheet.getLastRow();
  if (last < 2) return [];
  const vals = sheet
    .getRange(2, 1, last - 1, 1).getValues();

  const notes = [];
  vals.forEach(function (v, i) {
    const text = String(v[0]).trim();
    if (text && notes.length < NOTE_MAX) {
      notes.push({ row: i + 2, text: text });
    }
  });
  return notes;
}

// 한 칸을 3줄로 요약해 B열에도 저장합니다
function summarizeNote(row, text) {
  const out = SAM_SUMMARY(text);
  SpreadsheetApp.getActiveSpreadsheet()
    .getSheetByName(NOTE_TAB)
    .getRange(row, 2).setValue(out);
  return String(out);
}

// PDF를 저장하고 파일 이름과 링크를 돌려줍니다
function savePdfLink() {
  const file = savePdf();
  return { name: file.name, url: file.url };
}

  const all = ScriptApp.getProjectTriggers();
  for (const t of all) {

    }

  // 매일 8시대(8~9시 사이)에 mailList 실행

    .timeBased()
    .everyDays(1)
    .atHour(8)
    .create();
  Logger.log('트리거를 만들었습니다');
}

// 필요 권한: 스프레드시트 읽기·쓰기, 외부 연결
const MENU_NAME = 'AI 도우미';
const MENU_MAX_CELLS = 5;

// 시트를 열면 메뉴 두 개를 만듭니다
function onOpen() {
  SpreadsheetApp.getUi()
    .createMenu(MENU_NAME)
    .addItem('선택한 칸 요약',
      'summarizeSelection')
    .addItem('PDF 저장', 'savePdf')
    .addToUi();

// 고른 칸을 요약해 오른쪽 칸에 적습니다
function summarizeSelection() {
  const range = SpreadsheetApp
    .getActiveRange();
  if (range.getNumColumns() > 1) {
    throw new Error('한 열만 고르세요');
  }

  const cells = range.getValues();
  if (cells.length > MENU_MAX_CELLS) {
    throw new Error('5칸 이하로 고르세요');
  }

  cells.forEach(function (row, i) {
    const out = SAM_SUMMARY(row[0]);
    range.offset(i, 1, 1, 1).setValue(out);
  });
  Logger.log(cells.length + '칸을 요약했습니다');
}

// Folder.gs : 드라이브 폴더 만들기와 PDF 정리
// Pdf.gs 의 savePdf 로 만든 PDF를 옮깁니다
// 필요 권한: 구글 드라이브 파일·폴더 관리
const FOLDER_NAME = '수업 PDF';

// 폴더를 찾고, 없으면 새로 만듭니다
function makeFolder() {
  const found = DriveApp
    .getFoldersByName(FOLDER_NAME);
  if (found.hasNext()) {
    return found.next();
  }

  const folder = DriveApp
    .createFolder(FOLDER_NAME);
  Logger.log('폴더 생성: ' + folder.getUrl());
  return folder;
}

// 내 드라이브 첫 화면의 PDF를 폴더로 옮깁니다
function movePdfs() {
  const folder = makeFolder();
  const it = DriveApp.getRootFolder()
    .getFilesByType(MimeType.PDF);
  const files = [];
  while (it.hasNext()) files.push(it.next());

  files.forEach(function (f) {
    f.setName(stamp_(f.getName()));
    f.moveTo(folder);
  });
  Logger.log(files.length + '개 PDF를 옮겼습니다');
}

// 이름 앞에 오늘 날짜를 붙입니다(이미 있으면 그대로)
function stamp_(name) {
  const zone = Session.getScriptTimeZone();
  const day = Utilities.formatDate(
    new Date(), zone, 'yyyy-MM-dd');
  return name.indexOf(day) === 0
    ? name : day + '_' + name;
}
