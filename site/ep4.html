// Slides.gs : '대본' 탭 → 구글 슬라이드
const SHEET_NAME = '대본';
const MAX_SLIDES = 20;
const MAIN_COLOR = '#D6232B'; // 명세서 메인색

// 시트를 열면 '발표 도우미' 메뉴를 만듭니다
function onOpen() {
  SpreadsheetApp.getUi()
    .createMenu('발표 도우미')
    .addItem('슬라이드 만들기', 'makeSlides')
    .addToUi();
}

// 대본 탭의 줄마다 슬라이드를 한 장씩 만듭니다
function makeSlides() {
  const ss = SpreadsheetApp
    .getActiveSpreadsheet();
  const sh = ss.getSheetByName(SHEET_NAME);
  if (!sh) {
    throw new Error('대본 탭이 없습니다.');
  }
  const rows = sh.getDataRange().getValues();
  rows.shift(); // 1행(머리글)은 뺀다
  const items = rows.filter(r => r[0]);

  if (items.length === 0) {
    throw new Error('2행부터 대본을 적으세요.');
  }
  if (items.length > MAX_SLIDES) {
    throw new Error('20장 이하로 줄이세요.');
  }
  const deck = SlidesApp.create(
    '발표 초안 - ' + ss.getName());
  const blank = deck.getSlides()[0];
  items.forEach(r => addSlide(deck, r));

  blank.remove(); // 빈 첫 장 지우기
  const url = deck.getUrl();
  console.log('완성: ' + url);
  ss.toast('내 드라이브에 저장했습니다', '완성', 10);
  return url;
}

// 슬라이드 한 장에 제목·내용·발표자 노트를 채웁니다
function addSlide(deck, r) {
  const s = deck.appendSlide(SlidesApp
    .PredefinedLayout.TITLE_AND_BODY);
  const T = SlidesApp.PlaceholderType;

  const title = s.getPlaceholder(T.TITLE);
  const body = s.getPlaceholder(T.BODY);
  if (!title || !body) {
    throw new Error('자리표시자를 못 찾았습니다.');
  }
  const t = title.asShape().getText()
    .setText(String(r[0]));
  t.getTextStyle()
    .setForegroundColor(MAIN_COLOR);

  const text = String(r[1]).split('/')
    .map(v => v.trim()).join('\n');
  body.asShape().getText().setText(text);
  if (r[2]) {
    s.getNotesPage().getSpeakerNotesShape()
      .getText().setText(String(r[2]));
  }
}

// 대본 탭의 제목·핵심·발표할 말을 문서 한 개로 만듭니다
function makeScriptDoc() {
  const ss = SpreadsheetApp
    .getActiveSpreadsheet();
  const sh = ss.getSheetByName(SHEET_NAME);
  if (!sh) {
    throw new Error('대본 탭이 없습니다.');
  }
  const rows = sh.getDataRange().getValues();
  rows.shift(); // 1행(머리글)은 뺀다
  const items = rows.filter(r => r[0]);
  if (items.length === 0) {
    throw new Error('2행부터 대본을 적으세요.');
  }

  const doc = DocumentApp.create(
    '발표 대본 - ' + ss.getName());
  const body = doc.getBody();
  const H = DocumentApp.ParagraphHeading;
  const G = DocumentApp.GlyphType;
  body.getParagraphs()[0]
    .setText(ss.getName())
    .setHeading(H.TITLE);

  items.forEach((r, i) => {
    body.appendParagraph(
      (i + 1) + '. ' + r[0])
      .setHeading(H.HEADING2);
    pointsOf_(r[1]).forEach(p => {
      body.appendListItem(p)
        .setGlyphType(G.BULLET);
    });
    if (r[2]) {
      body.appendParagraph('발표할 말: ' + r[2]);
    }
  });

  const url = doc.getUrl();
  doc.saveAndClose();
  console.log('대본 완성: ' + url);
  return url;
}

// B열 글을 '/' 기준으로 잘라 핵심 문장 목록으로
function pointsOf_(cell) {
  return String(cell).split('/')
    .map(v => v.trim())
    .filter(v => v !== '');
}

// Timer.gs : 대본 글자 수로 발표 시간 계산
// Slides.gs 의 SHEET_NAME 이 먼저 필요합니다
// 필요 권한: 구글 스프레드시트 읽기
const CHARS_PER_MIN = 350; // 1분에 말하는 글자 수
const LIMIT_MIN = 0.5; // 제한 시간(분), 바꿔 쓰기

// 발표할 말(C열) 글자 수를 시간으로 바꿉니다
function estimateMinutes() {
  const sh = SpreadsheetApp
    .getActiveSpreadsheet()
    .getSheetByName(SHEET_NAME);
  if (!sh) {
    throw new Error('대본 탭이 없습니다.');
  }

  const rows = sh.getDataRange().getValues();
  rows.shift(); // 1행(머리글)은 뺀다
  const chars = sumChars_(rows);
  const min = Math.round(
    chars / CHARS_PER_MIN * 10) / 10;

  console.log(chars + '자 = 약 ' + min + '분');
  if (min > LIMIT_MIN) {
    console.log('경고: ' + LIMIT_MIN
      + '분을 넘었습니다.');
  }
  return min;
}

// 모든 줄의 발표할 말 글자 수를 더합니다
function sumChars_(rows) {
  let sum = 0;
  rows.forEach(r => {
    sum += String(r[2]).replace(/\s/g, '')
      .length;
  });
  return sum;
}

// Check.gs : 빈 칸·너무 긴 문장을 노란색으로 칠하기
// Slides.gs 의 SHEET_NAME 이 먼저 필요합니다
// 필요 권한: 구글 스프레드시트 읽기·쓰기
const CHECK_COLOR = '#FFF176'; // 노란색
const MAX_SENTENCE = 60; // 한 문장 최대 글자 수

// 대본 탭에서 고칠 칸을 찾아 노란색으로 칠합니다
function highlightProblems() {
  const sh = SpreadsheetApp
    .getActiveSpreadsheet()
    .getSheetByName(SHEET_NAME);
  if (!sh || sh.getLastRow() < 2) {
    throw new Error('2행부터 대본을 적으세요.');
  }

  const n = sh.getLastRow() - 1;
  const range = sh.getRange(2, 1, n, 3);
  range.setBackground(null); // 지난번 색을 지운다
  let bad = 0;

  range.getValues().forEach((row, i) => {
    row.forEach((cell, j) => {
      if (!isBad_(cell, row)) return;
      range.getCell(i + 1, j + 1)
        .setBackground(CHECK_COLOR);
      bad++;
    });
  });
  console.log('고칠 칸 ' + bad + '개를 칠했습니다.');
}

// 빈 칸이거나 한 문장이 너무 길면 true
function isBad_(cell, row) {
  const text = String(cell).trim();
  if (text === '') {
    return row.join('') !== '';
  }
  return text.split(/[.?!/]/).some(
    s => s.trim().length > MAX_SENTENCE);
}
